$ErrorActionPreference = 'Stop'

# 【R-3】Emit-Decision の二重出力防止フラグ。stdout へ判定 JSON を一度出したら、
# 何が起きても二度目は出さない（理由は Emit-Decision 定義部のコメント参照）。
$script:emitted = $false

# =============================================================================
# block-dangerous.ps1（修正版 / UTF-8 経路是正 + exit 2 ブロック）
#
# 【旧版の欠陥】
# 旧版は stdin を [Console]::In.ReadToEnd() で読んでいた。Windows PowerShell 5.1 の
# [Console]::InputEncoding は既定でコンソール入力コードページ（当環境では CP932）で
# あるため、Claude Code (Node.js) が UTF-8 で書いた JSON が CP932 として誤読される。
# 作業ディレクトリのパスに日本語（「ドキュメント」）を含む当環境では、
#   ト = E3 83 88 の 3 バイト目 0x88 が直後の "\" (0x5C) を 2 バイト文字の後半として
#   食う（CP932 のいわゆる「ダメ文字」）
# ため cwd の JSON エスケープが壊れ、ConvertFrom-Json が必ず失敗し、本フックは
# 全コマンドで fail-safe の ask に倒れていた（＝危険コマンド検知が常時無力化）。
#
# 【是正の三本柱】
# (1) stdin を [Console]::OpenStandardInput() + UTF-8 StreamReader で明示的に読む。
#     BOM 検出を有効にし（BOM 有無どちらでも壊れない）、不正バイトは例外ではなく
#     U+FFFD へ置換する。JSON の構造文字はすべて ASCII ゆえ、値の一部が化けても
#     判定を続行できる方が安全側である。
# (2) stdout へ [Console]::OpenStandardOutput() に生の UTF-8 バイトで直接書く。
#     PowerShell の出力パイプラインを経由しないので [Console]::OutputEncoding や
#     $OutputEncoding、PSVersion の差に一切左右されない。BOM は絶対に前置しない
#     （Claude Code は stdout の先頭文字が '{' でなければ JSON として解析せず、
#       判定が丸ごと黙殺されるため。BOM 前置は致命傷になる）。
# (3) deny は JSON に加えて stderr へ理由を書き、プロセスの終了コード 2 で返す
#     （終了コードの返し方自体は下記【F-1】参照。exit 文ではなく
#     [System.Environment]::Exit() を使う）。
#
# 【deny を終了コード2で返す理由】
# 公式ドキュメント（https://code.claude.com/docs/en/permissions の
# "Extend permissions with hooks" 節）に次の明記がある。
#   "A blocking hook also takes precedence over allow rules. A hook that exits with
#    code 2 stops the tool call before permission rules are evaluated, so the block
#    applies even when an allow rule would otherwise let the call proceed."
# 一方、JSON の permissionDecision だけの経路が allow ルールに優先するとは
# どこにも書かれていない。settings.json の allow に "Bash(git -C *)" のような
# 広い札があると、JSON 判定のみでは素通りする恐れが強い（実際、git -C . reset --hard
# が止まらなかった事象の最有力仮説がこれである）。確実に止めるには終了コード2が要る。
# ask は「止めずに確認させる」意図なので終了コード0のまま JSON で返す。
#
# 【F-1: exit 文ではなく [System.Environment]::Exit() を使う理由（重要・実機計測）】
# 信玄が実機で計測した事実（推論ではない）。同じ "exit 2" でも、フックの起動形に
# よって親プロセスに伝わる終了コードが変わる。
#
#   起動形                                      | "exit 2" の結果 | [Environment]::Exit(2) の結果
#   powershell.exe -File script.ps1              | 2               | 2
#   powershell.exe -Command '& "script.ps1"'     | 1               | 2
#   pwsh.exe        -Command '& "script.ps1"'    | 1               | 2
#
# 本番の ~/.claude/settings.json は
#   "command": "& \"$HOME\.claude\hooks\block-dangerous.ps1\"", "shell": "powershell"
# というアンパサンド呼び出し形（表の2行目）である。この形では PowerShell ホストが
# 子スクリプトの exit コードを正しく引き継がず、ホストプロセス自体の終了コードは
# 常に 1（一般エラー）に丸められる。exit 1 は「non-blocking error」と解釈され
# ツール呼び出しはそのまま進むため、deny のつもりが素通りに化け、今回の是正の
# 主目的（層2：終了コード2でallow札に打ち勝つ）が達成できない。
#
# そのため本ファイルはスクリプト内のすべての終了処理を [System.Environment]::Exit()
# に統一する。今後 "elegant に exit 文へ戻したい" という誘惑が来ても、上表の実測
# 結果があるかぎり戻してはならない。[System.Environment]::Exit() はプロセスを
# 即時終了させ、try/catch でも捕まらない（信玄が
# try { [System.Environment]::Exit(2) } catch { exit 0 } で実機検証し、
# 終了コード2になることを確認済み）。捕まらない以上、呼び出す前に stdout/stderr の
# バッファを必ず Flush 済みにしておく必要がある（下記 Write-RawUtf8 は書き込みの都度
# Flush() しているため現状で足りるが、出力ヘルパを書き換える際は要注意）。
#
# 【ファイルのエンコーディング】
# 本文に日本語リテラル（正規表現の Name や理由文）を含むため、このファイル自体は
# 必ず BOM 付き UTF-8 で保存すること。PS 5.1 は BOM なし UTF-8 の .ps1 を CP932 と
# して読むため、BOM を落とすと理由文が化けるだけでなく判定名も壊れる。
#
# 【意図的に避けた書き方】
# ・バッククォート（`r`n / `t / 行継続）は一切使わない。[char] 定数と
#   [System.Environment]::NewLine で代替する。生成・移送時に壊れにくくするため。
# ・pwsh 7 専用の構文（-Encoding utf8BOM 等）は使わない。5.1 と 7 の双方で動く
#   .NET API のみで構成する。
# =============================================================================

# --- 出力ヘルパ（すべて生バイト。PowerShell の出力エンコーディングに依存しない）---
function Write-RawUtf8($stream, [string]$text) {
  $bytes = [System.Text.Encoding]::UTF8.GetBytes($text)
  $stream.Write($bytes, 0, $bytes.Length)
  $stream.Flush()
}

function Write-StdoutUtf8([string]$text) {
  Write-RawUtf8 ([Console]::OpenStandardOutput()) $text
}

function Write-StderrUtf8([string]$text) {
  Write-RawUtf8 ([Console]::OpenStandardError()) $text
}

# fail-safe / deny の発生を後から追えるようにする監査ログ。
# 今回の事故は「フックが常時 fail-safe に倒れていたのに誰も気づけなかった」ことが
# 被害を長引かせた。その再発検知器である。ログ出力が失敗してもフック本体は止めない。
function Write-AuditLog([string]$line) {
  try {
    $dir = $env:TEMP
    if (-not $dir) { return }
    $path = Join-Path $dir 'claude-hook-block-dangerous.log'
    $stamp = (Get-Date).ToString('yyyy-MM-dd HH:mm:ss')
    $enc = New-Object System.Text.UTF8Encoding($false)
    $text = $stamp + [char]9 + $line + [System.Environment]::NewLine
    [System.IO.File]::AppendAllText($path, $text, $enc)
  } catch { }
}

# permissionDecision を stdout に出す共通ヘルパ。棄権路（fail-safe）からも呼べるよう
# try の外側・先頭で定義する。
#
# 【R-3】二重出力ガード：Deny-AndExit / Ask-AndExit は stdout へ JSON を書いた「後」に
# stderr 書き込みや監査ログ書き込みで例外が出る余地がある。旧構造ではその例外が
# 外側の catch に漏れ、catch が改めて Ask-AndExit を呼んで stdout に2つ目の JSON を
# 書いてしまっていた。stdout に JSON が2個並ぶと Claude Code 側は解析できず
# 「non-blocking error」＝ツールがそのまま進む、すなわち deny が素通りに化ける。
# ここで一度出したら二度と出さないようにして、その経路を断つ（対策の一枚目。
# 二枚目は Deny-AndExit / Ask-AndExit 自体を try/catch で包む対策。後述）。
function Emit-Decision($decision, $reason) {
  if ($script:emitted) { return }
  $script:emitted = $true
  $output = [ordered]@{
    hookSpecificOutput = [ordered]@{
      hookEventName = 'PreToolUse'
      permissionDecision = $decision
      permissionDecisionReason = $reason
    }
  }
  $json = $output | ConvertTo-Json -Depth 5 -Compress
  Write-StdoutUtf8 $json
}

# deny 専用。JSON を出したうえで stderr に理由を書き、監査ログに残し、確実に
# 終了コード2で止める。ドキュメント上、blocking message は「JSON の理由、無ければ
# stderr」の順で採られる。両方に出しておけば、どちらの経路で読まれても殿に理由が届く。
#
# 【R-3】stdout の JSON 出力（Emit-Decision）が終わった後段（stderr書き込み・監査ログ）
# で例外が出ても、ここで自前の try/catch に閉じ込めて外へ漏らさない。漏らすと
# 呼び出し元の catch が別の判定（Ask-AndExit 等）を再度呼び、二重出力の温床になる
# ためである。例外の有無にかかわらず、最後は必ず終了コード2で抜ける。
#
# 【F-1】exit ではなく [System.Environment]::Exit() を使う。理由はファイル冒頭の
# 【F-1】コメント（実機計測表）を参照。本番のアンパサンド呼び出し形では
# "exit 2" が終了コード1に化け、deny が素通りする。
function Deny-AndExit($reason) {
  try {
    Emit-Decision 'deny' $reason
    Write-StderrUtf8 $reason
    Write-AuditLog ('DENY: ' + $reason)
  } catch { }
  [System.Environment]::Exit(2)
}

# ask 専用。止める意図はないので終了コード0。fail-safe 由来のときだけ監査ログに残す。
# 【R-3】Deny-AndExit 同様、内部で例外が出ても外へ漏らさず、必ず終了コード0で抜ける。
# 【F-1】終了は [System.Environment]::Exit() で統一する（理由は冒頭コメント参照）。
function Ask-AndExit($reason, $isFailSafe) {
  try {
    Emit-Decision 'ask' $reason
    if ($isFailSafe) { Write-AuditLog ('FAILSAFE-ASK: ' + $reason) }
  } catch { }
  [System.Environment]::Exit(0)
}

# --- fail-safe 方針 ---
# 本フックは PreToolUse の matcher "Bash|PowerShell" にのみ発火する（settings.json で確認済み）。
# 両ツールとも tool_input.command を必ず持つため、command を取得できない／JSON を解釈できない／
# 評価中に例外が出る、はいずれも「異常事態」である。判定不能時は素通しでなく ask（殿に確認）へ
# 倒す（fail-safe）。deny ではなく ask なのは、フックが恒常故障しても運用が完全停止せず、
# 殿が都度判断して進めるようにするため。ただし今回の事故を踏まえ、fail-safe に落ちたことは
# 必ず監査ログに残し、沈黙のまま無力化されることを防ぐ。

$raw = $null
try {
  # [Console]::In は使わない（InputEncoding=CP932 で化けるため）。
  # 標準入力ストリームを直接開き、UTF-8 として読む。
  #   UTF8Encoding($false, $false) : BOM を出力しない / 不正バイトで例外を投げず U+FFFD へ置換
  #   StreamReader の第3引数 $true : 先頭の BOM を検出して読み飛ばす
  $stdinStream = [Console]::OpenStandardInput()
  $utf8Lenient = New-Object System.Text.UTF8Encoding($false, $false)
  $reader = New-Object System.IO.StreamReader($stdinStream, $utf8Lenient, $true)
  $raw = $reader.ReadToEnd()
  $reader.Dispose()
} catch {
  Ask-AndExit '標準入力を読み取れなかったため、安全側に倒して確認します（fail-safe）' $true
}

try {
  if ($null -eq $raw -or $raw.Trim().Length -eq 0) {
    Ask-AndExit '標準入力が空だったため、安全側に倒して確認します（fail-safe）' $true
  }
  # StreamReader の BOM 検出をすり抜けた U+FEFF や前後空白を落とす（二重の保険）。
  $trimChars = [char[]]@([char]0xFEFF, [char]0x200B, [char]32, [char]9, [char]13, [char]10)
  $raw = $raw.Trim($trimChars)
  $payload = ConvertFrom-Json $raw
} catch {
  Ask-AndExit '入力JSONを解析できなかったため、安全側に倒して確認します（fail-safe）' $true
}

try {
  # ペイロードが配列で来た場合に備えて先頭要素を取る。
  if ($payload -is [System.Array]) {
    if ($payload.Length -eq 0) {
      Ask-AndExit '入力JSONが空配列だったため、安全側に倒して確認します（fail-safe）' $true
    }
    $payload = $payload[0]
  }

  # Bash ツールも PowerShell ツールも tool_input.command にコマンド文字列を格納する。
  # 念のため他フィールド名（script/Command/Script）も保険として拾う。
  $command = $null
  foreach ($field in @('command', 'script', 'Command', 'Script')) {
    $value = $payload.tool_input.$field
    if ($value) {
      # 配列や数値で来ても -match が期待通り効くよう、必ず文字列にしてから扱う。
      if ($value -is [System.Array]) {
        $command = ($value -join ' ')
      } else {
        $command = [string]$value
      }
      break
    }
  }
  # Bash|PowerShell では command は必ず入る。取れないのは異常ゆえ ask に倒す。
  if ([string]::IsNullOrWhiteSpace($command)) {
    Ask-AndExit 'コマンド文字列を取得できなかったため、安全側に倒して確認します（fail-safe）' $true
  }

  # git の危険サブコマンド検知で、git 本体とサブコマンドの間に差し込まれても検知を外さないための
  # オプション挿入パターン。`-C <dir>`（作業ディレクトリ指定）や `-c <k=v>`（設定上書き）を
  # 任意個はさむ書き方（例: git -C . reset --hard）で deny を迂回されるのを塞ぐ。
  # 末尾に \s+ を含めることで、補間側は ${gitOpt} を副コマンドに直結でき（間にリテラル空白を
  # 置かない）、挿入が無いケース（git reset --hard）でも空マッチで正しく一致する。
  #
  # 【R-2】旧版は引数部を \S+（空白を含まない1トークン）で受けていたため、
  # git -C "My Folder" reset --hard のように引数を引用符で囲んで空白を含めると、
  # \S+ が引用符ごと壊れて $gitOpt 自体がマッチせず、全ての git 系 deny を回避できた。
  # 引数部を「二重引用符で囲む」「単一引用符で囲む」「空白を含まない生トークン」の
  # いずれかを受ける $gitOptArg に置き換えて塞ぐ。PowerShell のシングルクォート文字列
  # 内でリテラルの ' を書くには '' と重ねる必要があるため、正規表現中の単一引用符が
  # 二重に見えるが誤記ではない。
  $gitOptArg = '("[^"]*"|''[^'']*''|\S+)'

  # 【R-4】--no-pager / --git-dir= / --work-tree= / --config-env= も $gitOpt に
  # 差し込み可能なオプションとして許容し、これらを前置した検知回避を塞ぐ。
  # とくに --config-env=core.hooksPath=... は下の -c チェック（-c pager/alias/hooksPath）
  # を素通りする完全な代替迂回路であるため、必ず塞ぐ必要がある。
  $gitOpt = '(-C\s+' + $gitOptArg + '\s+' +
            '|-c\s+' + $gitOptArg + '\s+' +
            '|--no-pager\s+' +
            '|--git-dir=' + $gitOptArg + '\s+' +
            '|--work-tree=' + $gitOptArg + '\s+' +
            '|--config-env=' + $gitOptArg + '\s+' +
            ')*'

  # --- deny（ブロック）---
  $denyPatterns = @(
    @{ Name = 'rm -rf';                             Regex = 'rm\s+-rf' }
    @{ Name = 'git push --force/-f';                Regex = "git\s+${gitOpt}push\s+.*(--force|-f)\b" }
    @{ Name = 'git reset --hard';                   Regex = "git\s+${gitOpt}reset\s+--hard" }
    @{ Name = 'git clean -f';                       Regex = "git\s+${gitOpt}clean\s+-[a-z]*f" }
    @{ Name = 'git checkout -- (作業ツリー破棄)';   Regex = "git\s+${gitOpt}checkout\s+--\s" }
    @{ Name = 'git restore (作業ツリー破棄)';       Regex = "git\s+${gitOpt}restore\b" }
    @{ Name = 'フォーク爆弾';                        Regex = ':\(\)\s*\{' }
    @{ Name = 'mkfs';                               Regex = 'mkfs' }
    @{ Name = 'dd if=';                              Regex = 'dd\s+if=' }
    @{ Name = 'コマンド先頭の変数代入';              Regex = '^\s*[A-Za-z_]\w*\s*=' }
    @{ Name = '複合コマンド中の削除操作';            Regex = '[;&|]\s*(rm|rmdir|del|unlink|Remove-Item|Clear-Content)\b' }
    @{ Name = '複合コマンド中の危険操作';            Regex = '[;&|]\s*(sudo|chmod|chown|mkfs|dd)\b' }
    @{ Name = 'force push(+記法)';                  Regex = "git\s+${gitOpt}push\s+\S+\s+\+" }
    @{ Name = 'git commit -n(--no-verify略記)';     Regex = "git\s+${gitOpt}commit\s+-n\b" }
    # 【R-4】--no-verify の長綴り。既存の -n 略記の検知（直上）は残したまま、
    # 長い綴りも別途拾う。
    @{ Name = 'git commit --no-verify(長綴り)';     Regex = "git\s+${gitOpt}commit\b.*--no-verify\b" }
    # 【R-4】-nm のような結合短縮形（-n と他の短縮オプションを一つの - にまとめた形）を
    # 拾う。「-」の直後に2文字以上の英字が続き、その中に n を含むものにマッチさせる
    # ことで、単独の -n（既存パターンで検知済み）とは重複させずに結合形だけを拾う。
    # 旧設計案は commit 以降を .* で全域走査していたが、コミットメッセージの本文に
    # 偶然 "-no" 等の綴りが含まれるだけで誤検知（false positive で正当な commit を
    # 巻き込む）する恐れがあったため、既存の "commit -n\b" パターンと同じ流儀
    # （commit 直後から、他の "-" 始まりの短縮オプション群のみを挟んだ位置に限定）に
    # 絞った。この結果 "git commit -m \"msg\" -nm" のように引用符付き本文の後ろに
    # 結合形が来る書き方までは拾えない（既存の -n\b 単体パターンも同じ制約を持つ）。
    @{ Name = 'git commit -n結合短縮形(-nm等)';     Regex = "git\s+${gitOpt}commit\s+(-[a-zA-Z0-9=-]*\s+)*(-[a-zA-Z]*n[a-zA-Z]+|-[a-zA-Z]+n[a-zA-Z]*)\b" }
    @{ Name = 'find -exec/-delete';                 Regex = 'find\s+.*\s(-exec|-delete)\b' }
    @{ Name = '秘密ファイルへのアクセス(id_rsa/gh/aws等)'; Regex = 'id_rsa|\.credentials\.json|\.git-credentials|\.ssh[/\\]|\.config[/\\]gh[/\\]|hosts\.yml|\.aws[/\\]' }
    # 【R-4/残件4】ここから、高坂のハーネスで素通りが実測された・引継ぎ書第4節が
    # 挙げた git 危険操作。いずれも ${gitOpt} を挟んで -C 前置による回避を封じる。
    @{ Name = 'git filter-branch(履歴書き換え)';    Regex = "git\s+${gitOpt}filter-branch\b" }
    @{ Name = 'git rm(追跡ファイル削除。--cached含む)'; Regex = "git\s+${gitOpt}rm\b" }
    @{ Name = 'git mv(追跡ファイル移動)';           Regex = "git\s+${gitOpt}mv\b" }
    @{ Name = 'git checkout -f/--force(作業ツリー破棄)'; Regex = "git\s+${gitOpt}checkout\s+.*(-f\b|--force\b)" }
    @{ Name = 'git update-ref(参照の直接書き換え)'; Regex = "git\s+${gitOpt}update-ref\b" }
    # 【信玄実機監査・項目4】git difftool/mergetool の -x/--extcmd は、設定ファイルへの
    # 書き込みすら経ずに任意コマンドを直接起動できる（diff/mergeツールとして指定した
    # コマンドがそのまま実行される）。-x/--extcmd の前に \s を要求し、ファイル名等に
    # 偶然 "-x" が含まれるだけの誤検知（例: file-x.txt）を避ける。
    @{ Name = 'git difftool/mergetool -x/--extcmd(任意コマンド実行)'; Regex = "git\s+${gitOpt}(diff|merge)tool\b.*\s(-x\b|--extcmd=)" }
    # 【信玄実機監査・項目5】fetch/clone/pull/push の --upload-pack=/--receive-pack=/--exec=
    # は、ローカルの transport 起動時に指定コマンドをそのまま実行する（本来は
    # リモート側の git 実行ファイルパスを指す引数だが、任意文字列を渡せるため
    # 事実上の任意コマンド実行経路になる）。
    @{ Name = 'git fetch/clone/pull/push --upload-pack=/--receive-pack=/--exec=(任意コマンド実行)'; Regex = "git\s+${gitOpt}(fetch|clone|pull|push)\b.*\s(--upload-pack=|--receive-pack=|--exec=)" }
    # 【信玄実機監査・項目7】git remote set-url/add/remove/rename は push/fetch 先を
    # すり替えられ、他の防御をすべて素通りして攻撃者のリモートへ資材が流出しうる。
    # git remote -v（読み取り）・git remote（素）は対象外とする。-v 等のフラグは
    # サブコマンドの前にも置ける（例: git remote -v set-url origin <URL>）ため、
    # "remote\s+-v" のような先頭固定では取りこぼす。サブコマンド語が remote 以降の
    # どこに現れても拾えるよう .* で挟む。
    @{ Name = 'git remote set-url/add/remove/rename(push/fetch先の差し替え)'; Regex = "git\s+${gitOpt}remote\b.*\s(set-url|add|remove|rename)\b" }
  )

  foreach ($p in $denyPatterns) {
    if ($command -match $p.Regex) {
      Deny-AndExit "危険コマンド検知によりブロック: $($p.Name) [一致箇所: $($matches[0])]"
    }
  }

  # git -c による設定経由の任意実行。-C（大文字、ディレクトリ指定）を先に差し込んで
  # git -C . -c core.pager='...' の形で迂回されるのを塞ぐため、間に $gitOpt を許容する。
  # 末尾の -c だけは小文字限定にしたいので -cmatch（大文字小文字を区別）で判定する。
  #
  # 【R-1】固定したいのは「-c」という*オプション名*だけであり、その後に続く設定キー
  # 名（pager/alias/hooksPath）まで大文字小文字を固定する必要はない。git-config は
  # 変数名を大文字小文字非区別と定めており、core.PAGER や core.HooksPath のような
  # 綴りでも git は同じ意味で受理するため、キー名部分だけ固定していると
  # core.hookspath 等の綴り違いで deny を回避できてしまう。(?i:...) でキー名部分
  # だけを大文字小文字非区別にし、-c 自体（-C との区別）は -cmatch のまま維持する。
  if ($command -cmatch "git\s+${gitOpt}-c\s+\S*(?i:pager|alias|hooksPath)") {
    Deny-AndExit "危険コマンド検知によりブロック: gitを踏み台にした任意実行(-c pager/alias/hooksPath) [一致箇所: $($matches[0])]"
  }

  # 【信玄実機監査・項目1】--config-env=<name>=<envvar> は -c と全く同格の危険迂回路。
  # --config-env=core.pager=EVIL のように環境変数経由で core.pager 等を差し替えられ、
  # 以後の git log 等が任意コマンドを実行する状態を作れる。$gitOpt に --config-env= を
  # 「差し込まれても後続の危険サブコマンド検知を外さない」ために許容オプションとして
  # 加えてあるが、それだけでは --config-env 自体が危険という側面を塞げない
  # （--config-env を挟んだ先に危険サブコマンドが無いコマンド、例えば
  # "git --config-env=core.pager=EVIL log" は $gitOpt を通り抜けた後 log しか無く、
  # 既存の deny 群のどれにも一致しないため素通りしていた）。-c チェックと同じ流儀で
  # キー名部分だけ大文字小文字非区別にして独立に検知する。
  if ($command -cmatch "git\s+${gitOpt}--config-env=\S*(?i:pager|alias|hooksPath)") {
    Deny-AndExit "危険コマンド検知によりブロック: gitを踏み台にした任意実行(--config-env pager/alias/hooksPath) [一致箇所: $($matches[0])]"
  }

  # 【信玄実機監査・項目6】--exec-path=<dir> は git がサブコマンド実体（git-*
  # 実行ファイル）を探しに行くディレクトリそのものを差し替える。任意ディレクトリを
  # 指せば任意の実行ファイルを git のサブコマンドとして起動できるため、-c や
  # --config-env と同格に危険である。$gitOpt に足して後続コマンドの検知を通すだけでは
  # --exec-path 自体の危険性を見落とす（項目1と同じ誤りを繰り返さないため独立に置く）。
  if ($command -match "git\s+${gitOpt}--exec-path=") {
    Deny-AndExit "危険コマンド検知によりブロック: gitを踏み台にした任意実行(--exec-path) [一致箇所: $($matches[0])]"
  }

  # 【R-4/残件4】git branch -D（大文字D＝強制削除）。-d（小文字＝通常削除、マージ
  # 済みでないと失敗する安全な削除）は deny にしない。denyPatterns 配列は
  # -match（大文字小文字を区別しない）で回しているため、そこに素朴に足すと
  # -d まで巻き込んでしまう。ここだけ -cmatch で大文字小文字を区別して判定する。
  if ($command -cmatch "git\s+${gitOpt}branch\s+.*-D\b") {
    Deny-AndExit "危険コマンド検知によりブロック: git branch -D(強制削除) [一致箇所: $($matches[0])]"
  }

  # 【信玄裁定・項目2/3】git config は書き込み系（core.pager/core.hooksPath 等の
  # 差し替えを含む）だと、以後の無害な git log 等の実行時に時間差で任意コマンドが
  # 実行される状態を作れてしまう。危険が見えにくく ask で殿に判断を委ねるには
  # 情報が不足するため、信玄の裁定によりここは ask ではなく deny とする
  # （旧版は ask としていたが、信玄の実機監査結果を受けて deny に格上げした）。
  # 読み取り系（--get/--get-all/--list/-l）のみ除外する。--get-all は「--get」を
  # 前方一致で含むため実は既存の --get\b 除外だけでも塞げているが、明示のため
  # 別掲する。
  # deny への格上げに伴い、他の deny パターンより誤検知の重みが増す。"git config" は
  # コミットメッセージ等の地の文にも紛れ込みやすい語句（例: git commit -m
  # "update git config file" のような、値を書き込まない無害な commit）であるため、
  # このパターンだけは (^|[;&|]\s*) でコマンドの先頭またはコマンド区切り直後に
  # 出現する "git" だけを対象にし、引用符内の地の文に埋め込まれた "git config" を
  # 誤って deny しないようにする。
  if ($command -match "(^|[;&|]\s*)git\s+${gitOpt}config\s+(?!.*(--get\b|--get-all\b|--list\b|-l\b))") {
    Deny-AndExit "危険コマンド検知によりブロック: git config(値の書き込み。--get/--get-all/--list/-lは除外) [一致箇所: $($matches[0])]"
  }

  # --- ask（確認を挟む。ブロックはしない） ---
  $askPatterns = @(
    @{ Name = '秘密らしきファイル名'; Regex = '\.env\b|\.pem\b|credentials|secret|\.npmrc|\.aws[/\\]' }
    @{ Name = 'msedge直接起動(別綴り保険)'; Regex = 'msedge' }
    # git config の ask パターンは信玄の裁定により deny へ格上げした
    # （denyPatterns の直後・-c チェック付近を参照）。ここには残さない。
  )

  foreach ($p in $askPatterns) {
    if ($command -match $p.Regex) {
      Ask-AndExit "確認が必要なコマンド: $($p.Name) [一致箇所: $($matches[0])]" $false
    }
  }

  # settings.json / hooks / bin ディレクトリへの書き込み。deny にはせず ask とする
  # （update-config skill 等、正当な設定保守作業を止めないため）。
  # bin/ を加えたのは、bin/ が「書き込み自由＋python実行自由」の二重許可になっており、
  # 任意 .py を書いて実行する迂回路を断つため。書き込み動詞に New-Item/Add-Content/
  # Rename-Item/tee/touch を追加（旧版は New-Item 等が漏れており hooks/ すら New-Item -Force で
  # 無警告上書きできた）。
  # 旧版は行継続のバッククォートで 3 条件を繋いでいたが、移送時に壊れやすいので
  # 変数に分けた（判定内容は同一）。
  $hitSettings = $command -match '\.claude[/\\]settings.*\.json'
  $hitHooks = $command -match '\.claude[/\\]hooks[/\\]'
  $hitBin = $command -match '\.claude[/\\]bin[/\\]'
  $touchesProtectedPath = ($hitSettings -or $hitHooks -or $hitBin)
  $hasWriteVerb = $command -match '(>|Set-Content|Add-Content|Out-File|New-Item|Copy-Item|Move-Item|Rename-Item|\bcp\b|\bmv\b|\btee\b|\btouch\b|\becho\b)'
  if ($touchesProtectedPath -and $hasWriteVerb) {
    Ask-AndExit '設定ファイル/フック/binディレクトリへの書き込み操作のため確認が必要です' $false
  }

  # 【F-1】無害と判定できたときの正常終了も exit ではなく
  # [System.Environment]::Exit() に統一する（理由はファイル冒頭の【F-1】コメント参照）。
  [System.Environment]::Exit(0)
} catch {
  Ask-AndExit '検査中に例外が発生したため、安全側に倒して確認します（fail-safe）' $true
}
